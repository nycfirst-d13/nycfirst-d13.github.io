# Game Gallery Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the NYC FIRST D13 8-bit arcade game gallery — a Next.js static export that reads an approved-games Google Sheet CSV at runtime and shows a pixel-art grid plus a per-game iframe page.

**Architecture:** Two static client-rendered pages (`/` grid, `/games/` detail). Both fetch the same published Google Sheet CSV in the browser. The detail page reads the game id from `window.location.search` (`?id=`) — no `[id]` dynamic route, so new approved games go live without a rebuild. 8bitcn pixel components are vendored via the shadcn CLI. All heavy content (games, thumbnails, archives) lives off-repo (MakeCode + Google Drive + Sheets).

**Tech Stack:** Next.js (App Router, `output: 'export'`), TypeScript, Tailwind CSS, 8bitcn.com components (Radix-based), Papaparse (CSV), Vitest (unit tests), `next/font` (Press Start 2P + Geist).

## Global Constraints

- **Static only:** `output: 'export'`, `basePath: '/game-gallery'`, `trailingSlash: true`. No server, no API routes, no `generateStaticParams`.
- **No build-time dependence on game data:** game list/identity resolved at runtime in the browser. Never fetch the CSV at build time.
- **Commits:** repo root is the parent dir. Always `git -C /Users/avigoldman/Desktop/nycfirst-d13.github.io add game-gallery/<path>` then commit from there. No nested git repo.
- **Fonts:** Press Start 2P for headings ONLY; VT323 for all body/label text, base size **≥18px** (readability floor for grades 3–8). Both via `next/font/google`.
- **Roles/grades (exact, ordered):** `3, 4, 5, 6, 7, 8, Intern, Instructor`.
- **Palette (CSS vars, exact):** `--bg:#0A0E1A; --surface:#141929; --border:#1E2D4A; --accent:#E0241B; --accent-hi:#FF3B30; --blue:#2563EB; --ink:#F0F4FF; --ink-2:#9DA8C4;`
- **No live iframes on the grid** (thumbnails only). Single iframe on the detail page, preserving MakeCode 160:120 aspect ratio.
- **A11y:** ≥44px tap targets, `alt` on all thumbnails, visible focus states.
- **Thumbnail URL:** `https://makecode.com/api/{shareId}/thumb`, with `onError` → local placeholder.
- **CSV source:** `process.env.NEXT_PUBLIC_GAMES_CSV_URL` — the Google Sheet "Approved" tab published-to-web CSV link. Set in `game-gallery/.env.local` (and as a repo secret / Actions env for CI). Approved columns: `id, game_title, student_name, grade, student_url, d13_url, submitted_at, session`.

```ts
// game-gallery/lib/games.ts
import Papa from 'papaparse'

export type Game = {
  id: string
  game_title: string
  student_name: string
  grade: string
  student_url: string
  d13_url: string
  submitted_at: string
  session: string
}

// ponytail: Papaparse over a hand-rolled split — game titles contain commas; correct quote handling is not worth re-deriving.
export function parseGamesCsv(csv: string): Game[] {
  const { data } = Papa.parse<Record<string, string>>(csv, {
    header: true,
    skipEmptyLines: true,
    transform: (v) => v.trim(),
  })
  return data
    .map((r) => ({
      id: r.id ?? '',
      game_title: r.game_title ?? '',
      student_name: r.student_name ?? '',
      grade: r.grade ?? '',
      student_url: r.student_url ?? '',
      d13_url: r.d13_url ?? '',
      submitted_at: r.submitted_at ?? '',
      session: r.session ?? '',
    }))
    .filter((g) => g.id !== '')
}

export function sortByNewest(games: Game[]): Game[] {
  return [...games].sort((a, b) => (b.submitted_at || '').localeCompare(a.submitted_at || ''))
}

export function findGame(games: Game[], id: string): Game | undefined {
  const key = id.trim().toLowerCase()
  return games.find((g) => g.id.trim().toLowerCase() === key)
}

export function extractShareId(d13Url: string): string | null {
  try {
    const u = new URL(d13Url)
    const seg = u.pathname.split('/').filter(Boolean)
    return seg.length ? seg[seg.length - 1] : null
  } catch {
    return null
  }
}

export function thumbUrl(shareId: string): string {
  return `https://makecode.com/api/${shareId}/thumb`
}

export function gradeBadge(grade: string): { label: string; tone: 'blue' | 'intern' | 'instructor' } {
  const g = grade.trim()
  if (g === 'Instructor') return { label: g, tone: 'instructor' }
  if (g === 'Intern') return { label: g, tone: 'intern' }
  return { label: g, tone: 'blue' }
}

export async function fetchGames(csvUrl: string): Promise<Game[]> {
  const res = await fetch(csvUrl)
  if (!res.ok) throw new Error(`CSV fetch failed: ${res.status}`)
  return sortByNewest(parseGamesCsv(await res.text()))
}
```

```tsx
'use client'
import { useState } from 'react'
import { extractShareId, thumbUrl } from '@/lib/games'

const PLACEHOLDER = '/game-gallery/placeholder.svg' // basePath-prefixed

export function Thumb({ d13Url, alt }: { d13Url: string; alt: string }) {
  const shareId = extractShareId(d13Url)
  const [src, setSrc] = useState(shareId ? thumbUrl(shareId) : PLACEHOLDER)
  return (
    <img
      src={src}
      alt={alt}
      width={160}
      height={120}
      onError={() => { if (src !== PLACEHOLDER) setSrc(PLACEHOLDER) }}
      className="w-full image-rendering-pixelated"
      style={{ imageRendering: 'pixelated', aspectRatio: '160 / 120', objectFit: 'cover' }}
    />
  )
}
```

```tsx
'use client'
import { useEffect, useState } from 'react'
import { RetroModeSwitcher } from '@/components/ui/retro-mode-switcher'

export function CrtToggle() {
  const [on, setOn] = useState(false)

  useEffect(() => {
    const saved = localStorage.getItem('crt') === '1'
    setOn(saved)
    document.documentElement.classList.toggle('crt', saved)
  }, [])

  function toggle() {
    const next = !on
    setOn(next)
    document.documentElement.classList.toggle('crt', next)
    localStorage.setItem('crt', next ? '1' : '0')
  }

  return <RetroModeSwitcher checked={on} onClick={toggle} aria-label="Toggle CRT effect" />
}
```
